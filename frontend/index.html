const r = await fetch("https://api.example.com/v1/verifications", {
  method: "POST",
  headers: { Authorization: `Bearer ${process.env.API_KEY}` },
  body: JSON.stringify({ external_user_id: user.id })
});
const { telegram_url, id } = await r.json();
// redirect user → telegram_url
const iv = setInterval(async () => {
  const s = await fetch(`/v1/verifications/${id}`, { headers }).then(r=>r.json());
  if (s.status === "verified") clearInterval(iv);
}, 3000);
